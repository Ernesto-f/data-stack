def get_conn():
    sql_conn = os.getenv("UPLOAD_SQL_CONN_STR")
    if not sql_conn:
        raise RuntimeError("UPLOAD_SQL_CONN_STR não definido no ambiente do Airflow.")
    return pyodbc.connect(sql_conn)


s3 = boto3.client(
    "s3",
    endpoint_url=S3_ENDPOINT,
    aws_access_key_id=S3_ACCESS_KEY,
    aws_secret_access_key=S3_SECRET_KEY,
)

_valid_ident = re.compile(r"^[A-Za-z0-9_]+$")


def _validate_ident(x: str, label: str) -> str:
    x = (x or "").strip()
    if not _valid_ident.fullmatch(x):
        raise Exception(f"{label} inválido: '{x}'. Use apenas letras, números e underscore.")
    return x


def normalize_target(target_schema: str | None, target_table: str) -> tuple[str, dict]:
    """
    Retorna:
      - target_sql: [db].[schema].[table] OU [schema].[table]
      - parts: dict com db/schema/table (db pode ser None no modo 2-part)
    """
    t = (target_table or "").strip()

    if t.count(".") == 2:
        db, schema, table = t.split(".")
        db = _validate_ident(db, "Database")
        schema = _validate_ident(schema, "Schema")
        table = _validate_ident(table, "Table")
        return f"[{db}].[{schema}].[{table}]", {"db": db, "schema": schema, "table": table}

    if not target_schema:
        raise Exception("target_schema é obrigatório quando target_table não está no formato db.schema.table")

    schema = _validate_ident(target_schema, "Schema")
    table = _validate_ident(t, "Table")
    return f"[{schema}].[{table}]", {"db": None, "schema": schema, "table": table}


def parse_expected_columns(expected_columns: str | None) -> list[str] | None:
    if not expected_columns:
        return None
    cols = [c.strip() for c in expected_columns.split(",")]
    cols = [c for c in cols if c]
    return cols or None


def filter_df_to_expected(df: pd.DataFrame, expected: list[str] | None) -> pd.DataFrame:
    """
    Mantém somente as colunas esperadas (todas as linhas).
    Match case-insensitive.
    Erra se faltar coluna.
    """
    if expected is None:
        return df
    if df is None or df.empty:
        return df

    colmap = {str(c).strip().lower(): c for c in df.columns}

    missing = []
    real_cols = []
    for c in expected:
        key = c.strip().lower()
        if key not in colmap:
            missing.append(c)
        else:
            real_cols.append(colmap[key])

    if missing:
        raise Exception(f"Colunas faltando no arquivo (após map): {missing}")

    return df[real_cols]


def apply_column_mapping_manual(df: pd.DataFrame, column_map_json: str | None) -> pd.DataFrame:
    """
    column_map_json: {"Coluna no arquivo": "coluna_sql", ...}

    - exige correspondência exata do lado esquerdo com o header do arquivo
    - mantém só as colunas mapeadas
    - renomeia para o nome exato da coluna SQL (lado direito)
    """
    if not column_map_json:
        return df

    try:
        mapping = json.loads(column_map_json)
    except Exception as e:
        raise Exception(f"column_map_json inválido: {e}")

    if not isinstance(mapping, dict) or not mapping:
        raise Exception("column_map_json deve ser um JSON object não-vazio: {source: target}")

    missing = [src for src in mapping.keys() if src not in df.columns]
    if missing:
        raise Exception(f"Colunas do arquivo não encontradas no map: {missing}")

    df = df[list(mapping.keys())].copy()
    df.rename(columns=mapping, inplace=True)
    return df


def ensure_ingest_col_exists(cn, parts: dict):
    """
    Confirma se a tabela destino tem a coluna upload_history_id.
    Consulta no DB correto quando rota é 3-part.
    """
    db = parts["db"]
    schema = parts["schema"]
    table = parts["table"]

    cur = cn.cursor()

    if db:
        sql = f"""
        SELECT 1
          FROM [{db}].INFORMATION_SCHEMA.COLUMNS
         WHERE TABLE_SCHEMA = ?
           AND TABLE_NAME = ?
           AND COLUMN_NAME = ?
        """
        row = cur.execute(sql, schema, table, INGEST_COL).fetchone()
    else:
        sql = """
        SELECT 1
          FROM INFORMATION_SCHEMA.COLUMNS
         WHERE TABLE_SCHEMA = ?
           AND TABLE_NAME = ?
           AND COLUMN_NAME = ?
        """
        row = cur.execute(sql, schema, table, INGEST_COL).fetchone()

    if not row:
        prefix = f"{db}." if db else ""
        raise Exception(
            f"A tabela destino não possui a coluna '{INGEST_COL}'. "
            f"Adicione: ALTER TABLE {prefix}{schema}.{table} "
            f"ADD {INGEST_COL} BIGINT NULL;"
        )


def get_table_schema(cn, parts: dict) -> dict:
    """
    Retorna dict com o schema da tabela destino.
    """
    db = parts["db"]
    schema = parts["schema"]
    table = parts["table"]

    cur = cn.cursor()

    if db:
        sql = f"""
        SELECT
            COLUMN_NAME,
            DATA_TYPE,
            CHARACTER_MAXIMUM_LENGTH,
            NUMERIC_PRECISION,
            NUMERIC_SCALE
        FROM [{db}].INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_SCHEMA = ?
          AND TABLE_NAME = ?
        """
        rows = cur.execute(sql, schema, table).fetchall()
    else:
        sql = """
        SELECT
            COLUMN_NAME,
            DATA_TYPE,
            CHARACTER_MAXIMUM_LENGTH,
            NUMERIC_PRECISION,
            NUMERIC_SCALE
        FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_SCHEMA = ?
          AND TABLE_NAME = ?
        """
        rows = cur.execute(sql, schema, table).fetchall()

    result = {}
    for r in rows:
        result[r.COLUMN_NAME] = {
            "data_type": (r.DATA_TYPE or "").lower(),
            "max_length": r.CHARACTER_MAXIMUM_LENGTH,
            "precision": r.NUMERIC_PRECISION,
            "scale": r.NUMERIC_SCALE,
        }
    return result


def normalize_empty_tokens(df: pd.DataFrame) -> pd.DataFrame:
    """
    Normaliza placeholders comuns de Excel/CSV para None.
    """
    df = df.copy()

    empty_tokens = {
        "", "-", "--", "---",
        "N/A", "n/a", "NA", "na",
        "NULL", "null",
        "None", "none",
        "nan", "NaN"
    }

    def clean_value(x):
        if pd.isna(x):
            return None
        if isinstance(x, str):
            v = x.strip()
            if v in empty_tokens:
                return None
            return v
        return x

    for col in df.columns:
        df[col] = df[col].apply(clean_value)

    return df


def normalize_df_to_sql_types(cn, parts: dict, df: pd.DataFrame) -> pd.DataFrame:
    """
    Converte o DataFrame de acordo com o tipo real da tabela SQL Server.
    """
    df = normalize_empty_tokens(df)
    df = df.copy()

    schema_info = get_table_schema(cn, parts)

    for col in df.columns:
        if col not in schema_info:
            continue

        sql_type = schema_info[col]["data_type"]
        scale = schema_info[col]["scale"]

        # TEXTOS
        if sql_type in ("varchar", "nvarchar", "char", "nchar", "text", "ntext"):
            df[col] = df[col].apply(lambda x: None if x is None else str(x))

        # INTEIROS
        elif sql_type in ("bigint", "int", "smallint", "tinyint"):
            df[col] = pd.to_numeric(df[col], errors="coerce")
            df[col] = df[col].apply(lambda x: None if pd.isna(x) else int(x))

        # DECIMAL / NUMERIC / MONEY
        elif sql_type in ("decimal", "numeric", "money", "smallmoney"):
            df[col] = pd.to_numeric(df[col], errors="coerce")

            def conv_decimal(x):
                if pd.isna(x):
                    return None
                xf = float(x)
                if not math.isfinite(xf):
                    return None

                if scale is not None:
                    q = Decimal("1." + ("0" * int(scale)))
                    return Decimal(str(xf)).quantize(q, rounding=ROUND_HALF_UP)
                return Decimal(str(xf))

            df[col] = df[col].apply(conv_decimal)

        # FLOAT / REAL
        elif sql_type in ("float", "real"):
            df[col] = pd.to_numeric(df[col], errors="coerce")
            df[col] = df[col].apply(
                lambda x: None if pd.isna(x) or not math.isfinite(float(x)) else float(x)
            )

        # DATAS
        elif sql_type in ("date", "datetime", "datetime2", "smalldatetime", "datetimeoffset"):
            df[col] = pd.to_datetime(df[col], errors="coerce")
            df[col] = df[col].astype(object)
            df[col] = df[col].apply(
                lambda x: None if pd.isna(x) else x.to_pydatetime()
            )

        # BIT
        elif sql_type == "bit":
            def conv_bit(x):
                if x is None:
                    return None
                if isinstance(x, str):
                    v = x.strip().lower()
                    if v in ("1", "true", "sim", "yes", "y"):
                        return True
                    if v in ("0", "false", "nao", "não", "no", "n"):
                        return False
                    return None
                return bool(x)

            df[col] = df[col].apply(conv_bit)

        # UNIQUEIDENTIFIER
        elif sql_type == "uniqueidentifier":
            df[col] = df[col].apply(lambda x: None if x is None else str(x))

        # FALLBACK
        else:
            df[col] = df[col].apply(lambda x: None if pd.isna(x) else x)

    return df


def insert_dataframe(cn, target_sql: str, parts: dict, df: pd.DataFrame) -> int:
    if df is None or df.empty:
        return 0

    df = normalize_df_to_sql_types(cn, parts, df)

    
    df = df.astype(object).where(pd.notna(df), None)

    cols = list(df.columns)
    if not cols:
        return 0

    print("==================================================")
    print("TABELA DESTINO:", target_sql)
    print("COLUNAS DF:", cols)
    print("QTD LINHAS:", len(df))
    print("DTYPES:")
    print(df.dtypes)

    col_list = ",".join([f"[{c}]" for c in cols])
    placeholders = ",".join(["?"] * len(cols))
    sql = f"INSERT INTO {target_sql} ({col_list}) VALUES ({placeholders})"

    records = [tuple(row) for row in df.itertuples(index=False, name=None)]

    print("==== TIPOS PYTHON DA PRIMEIRA LINHA ====")
    if records:
        first = records[0]
        for i, v in enumerate(first):
            print(i, cols[i], type(v), repr(v))
    print("========================================")

    cur = cn.cursor()
    cur.fast_executemany = True
    cur.executemany(sql, records)
    cn.commit()
    return len(records)


def override_delete_all(cn, target_sql: str):
    """
    OVERRIDE: apaga tudo antes de inserir.
    (DELETE é mais compatível que TRUNCATE.)
    """
    cur = cn.cursor()
    cur.execute(f"DELETE FROM {target_sql};")
    cn.commit()


def compute_id_range_for_upload(
    cn,
    target_sql: str,
    id_column_name: str,
    upload_history_id: int,
) -> tuple[int | None, int | None]:
    """
    Retorna (min, max) do id_column_name para linhas com upload_history_id = upload_history_id.
    """
    id_col = _validate_ident(id_column_name, "id_column_name")
    cur = cn.cursor()

    row = cur.execute(
        f"SELECT MIN([{id_col}]) AS id_min, MAX([{id_col}]) AS id_max "
        f"FROM {target_sql} WHERE [{INGEST_COL}] = ?",
        upload_history_id
    ).fetchone()

    if not row:
        return None, None

    return row.id_min, row.id_max


# =============================================================================
# DAG
# =============================================================================
with DAG(
    dag_id="file_upload_job",
    schedule=None,
    start_date=datetime(2025, 1, 1),
    catchup=False,
    tags=["fileupload"],
) as dag:

    @task
    def process(**ctx):
        conf = ctx["dag_run"].conf or {}

        upload_id = conf["upload_id"]
        upload_history_id = conf["upload_history_id"]
        route_key = conf["route_key"]
        s3_key = conf["s3_key"]

        tmp_path = None
        total_inserted = 0

        try:
            # 1) PROCESSING
            with get_conn() as cn:
                cn.cursor().execute(
                    """
                    UPDATE control.tb_file_upload_history
                       SET status = 'PROCESSING',
                           processing_start_at = SYSDATETIME()
                     WHERE upload_id = ?
                    """,
                    upload_id,
                )
                cn.commit()

            # 2) Resolve rota
            with get_conn() as cn:
                route = cn.cursor().execute(
                    """
                    SELECT target_schema,
                           target_table,
                           sheet_name,
                           expected_columns,
                           load_mode,
                           id_column_name,
                           column_map_json
                      FROM control.tb_file_route_upload
                     WHERE route_key = ?
                    """,
                    route_key,
                ).fetchone()

                if not route:
                    raise Exception("route_key não encontrada em control.tb_file_route_upload")

                target_schema = route.target_schema
                target_table = route.target_table
                sheet_name = route.sheet_name
                expected_columns = route.expected_columns
                load_mode = (route.load_mode or "INCREMENTAL").strip().upper()
                id_column_name = route.id_column_name
                column_map_json = route.column_map_json

            if load_mode not in ("INCREMENTAL", "OVERRIDE"):
                raise Exception(f"load_mode inválido: '{load_mode}'. Use INCREMENTAL ou OVERRIDE.")

            expected = parse_expected_columns(expected_columns)
            target_sql, parts = normalize_target(target_schema, target_table)

            # 3) Confirma upload_history_id na tabela destino
            with get_conn() as cn:
                ensure_ingest_col_exists(cn, parts)

            print(f"[DEBUG] S3_ENDPOINT={S3_ENDPOINT}")
            print(f"[DEBUG] BUCKET={BUCKET}")
            print(f"[DEBUG] s3_key={s3_key}")
            print(f"[DEBUG] upload_id={upload_id}")
            print(f"[DEBUG] upload_history_id={upload_history_id}")
            print(f"[DEBUG] S3_ACCESS_KEY={S3_ACCESS_KEY}")
            print(f"[DEBUG] S3_SECRET_KEY set?={'YES' if S3_SECRET_KEY else 'NO'}")

            # 4) Download MinIO
            with tempfile.NamedTemporaryFile(delete=False) as tmp:
                s3.download_fileobj(BUCKET, s3_key, tmp)
                tmp_path = tmp.name

            sk = (s3_key or "").lower()

            # 5) OVERRIDE: apaga tudo antes de inserir
            if load_mode == "OVERRIDE":
                with get_conn() as cn:
                    override_delete_all(cn, target_sql)

            # 6) Ingest
            if sk.endswith(".csv"):
                for chunk in pd.read_csv(tmp_path, chunksize=50_000):
                    chunk = apply_column_mapping_manual(chunk, column_map_json)
                    chunk = filter_df_to_expected(chunk, expected)
                    chunk[INGEST_COL] = int(upload_history_id)

                    with get_conn() as cn:
                        total_inserted += insert_dataframe(cn, target_sql, parts, chunk)

            elif sk.endswith(".xlsx"):
                if sheet_name:
                    df = pd.read_excel(tmp_path, sheet_name=sheet_name)
                else:
                    df = pd.read_excel(tmp_path)

                df = apply_column_mapping_manual(df, column_map_json)
                df = filter_df_to_expected(df, expected)
                df[INGEST_COL] = int(upload_history_id)

                with get_conn() as cn:
                    total_inserted += insert_dataframe(cn, target_sql, parts, df)

            else:
                raise Exception("Formato não suportado (somente .csv ou .xlsx)")

            # 7) Calcula range (se id_column_name configurado)
            id_min = None
            id_max = None
            id_col_name = None

            if id_column_name:
                id_col_name = id_column_name.strip()
                with get_conn() as cn:
                    id_min, id_max = compute_id_range_for_upload(
                        cn,
                        target_sql,
                        id_col_name,
                        upload_history_id,
                    )

            # 8) DONE + range
            with get_conn() as cn:
                cn.cursor().execute(
                    """
                    UPDATE control.tb_file_upload_history
                       SET status = 'DONE',
                           rows_total = ?,
                           rows_valid = ?,
                           id_col_name = ?,
                           id_min = ?,
                           id_max = ?,
                           processing_end_at = SYSDATETIME()
                     WHERE upload_id = ?
                    """,
                    total_inserted,
                    total_inserted,
                    id_col_name,
                    id_min,
                    id_max,
                    upload_id,
                )
                cn.commit()

        except Exception as e:
            with get_conn() as cn:
                cn.cursor().execute(
                    """
                    UPDATE control.tb_file_upload_history
                       SET status = 'FAILED',
                           error_message = ?,
                           processing_end_at = SYSDATETIME()
                     WHERE upload_id = ?
                    """,
                    str(e),
                    upload_id,
                )
                cn.commit()
            raise

        finally:
            if tmp_path and os.path.exists(tmp_path):
                os.remove(tmp_path)

    process()
