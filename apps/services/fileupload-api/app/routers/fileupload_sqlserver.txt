import uuid
from fastapi import APIRouter, UploadFile, File, Form, HTTPException

from app.core.minios3 import upload_fileobj
from app.core.config import settings
from app.core.sqlserver import get_route, insert_log, get_status
from app.core.airflow import trigger

router = APIRouter(prefix="/upload", tags=["fileupload"])



# EXTRAI A EXTESÃO DO ARQUIVO
def _ext_of(filename: str) -> str:
    if not filename or "." not in filename:
        return ""
    return filename.rsplit(".", 1)[-1].lower().strip()


@router.post("/")
async def fileupload(
    route_key: str = Form(...), 
    file: UploadFile = File(...)
    ):
    route_key = route_key.strip().upper()

    route = get_route(route_key)
    if not route:
        raise HTTPException(404, "route_key inválida ou inativa")

    filename = file.filename or "upload"
    ext = _ext_of(filename)

    allowed = (route.allowed_ext or "xlsx").lower().replace(" ", "")
    allowed_set = {x for x in allowed.split(",") if x}

    if ext not in allowed_set:
        raise HTTPException(400, f"Extensão não permitida: .{ext}. Permitidas: {sorted(allowed_set)}")

    upload_id = str(uuid.uuid4())
    s3_key = f"uploads/{route_key}/{upload_id}__{filename}"

    # 1) salva no MinIO (S3)
    upload_fileobj(
        bucket=settings.S3_BUCKET,
        key=s3_key,
        fileobj=file.file,
        content_type=file.content_type,
        )

    # 2) grava histórico
    upload_history_id = insert_log(
            upload_id=upload_id,
            route_key=route_key,
            file_name=filename,
            file_size_bytes=getattr(file, "size", None) or 0,
            bucket=settings.S3_BUCKET,
            s3_key=s3_key,
            uploaded_by=None,
            source_system="portal-fileupload",
    )

    # 3) dispara Airflow
    dag_run_id = trigger(upload_id, upload_history_id, route_key, s3_key)

    return {
    "upload_id": upload_id,
    "upload_history_id": upload_history_id,
    "dag_run_id": dag_run_id,
    }


@router.get("/status/{upload_id}")
def status(upload_id: str):
    row = get_status(upload_id)
    if not row:
        return {"status": "NOT_FOUND"}
    return {"status": row.status, "error": row.error_message}
